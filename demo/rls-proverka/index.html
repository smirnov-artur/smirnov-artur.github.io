Как проверить, что клуб А не видит данные клуба Б, а федерация видит только сводку

Метод и готовые SQL-проверки. Прогон — на модели по описанию проекта (федерация → клубы → игроки, рейтинги, турниры, планы развития), это не ваш код. Окружение как в Supabase: роли anon и authenticated, auth.uid() из JWT. База — PostgreSQL 17.5.

6 → 0дыр до исправлений и после
7 → 3находок статики (после остались только «сверить логику» у функций)

Метод

  1. Статика по схеме (audit.sql, хватает выгрузки схемы без данных): таблицы без RLS, политики «всегда да», представления без security_invoker — они читают таблицы правами владельца и обходят RLS, таблицы с данными игроков без org_id, функции SECURITY DEFINER, доступные через rpc.
  2. Живые проверки под пользователями (izolyaciya.sql): роль authenticated и sub в JWT конкретного тренера клуба А, тренера клуба Б, аналитика федерации и не вошедшего. Каждый пытается прочитать, добавить, изменить и «переманить» чужое. Все записи откатываются, база не меняется.
  3. Федерация: сводка — отдельная функция только с агрегатами по клубам своей федерации. Проверяется и то, что сводка по всем клубам есть, и то, что ФИО и телефонов игроков федерации не видно, и то, что тренеру клуба сводка не отдаётся.
  4. Исправления миграцией (fixes.sql) и повторный прогон: все проверки зелёные.
  5. Проверки — в CI: прогон на каждый pull request с изменением схемы, новая таблица без политики не уедет в прод.

Прогон до исправлений

audit.sql — Статика по схеме: таблицы без RLS, политики «всегда да», представления в обход RLS, таблицы без org_id, функции с правами владельца
-- Статическая проверка RLS по схеме (выгрузка схемы без данных или копия проекта). Запуск: SQL Editor / psql, роль postgres.
-- Одна таблица на выходе: что проверяли, где нашли, чем грозит. Пусто — по этим пунктам чисто.
with
tabl as (
  select c.oid, c.relname as t, c.relrowsecurity as rls
  from pg_class c join pg_namespace n on n.oid = c.relnamespace
  where n.nspname = 'public' and c.relkind in ('r', 'p')
),
pol as (select * from pg_policies where schemaname = 'public')
select * from (
  -- 1. Таблица в открытой схеме без RLS: через API её читает и меняет любой вошедший (а с грантом anon — и не вошедший)
  select 1 as n, 'RLS выключен' as proverka, t as obekt, 'таблицу видят все пользователи всех клубов' as chem_grozit
  from tabl where not rls
  union all
  -- 2. RLS включён, а политик нет — для пользователей таблица пустая (часто так «чинят» и ломают экран)
  select 2, 'RLS без политик', t, 'пользователи не видят ничего; проверить, так ли задумано'
  from tabl where rls and not exists (select 1 from pol where pol.tablename = tabl.t)
  union all
  -- 3. Политика «всегда да» на чтение или запись
  select 3, 'политика всегда пропускает', tablename || '.' || policyname || ' (' || cmd || ')',
         case when cmd in ('INSERT', 'UPDATE', 'ALL') and coalesce(with_check, '') = 'true'
              then 'можно записать строку в чужой клуб' else 'видны строки всех клубов' end
  from pol where coalesce(qual, '') = 'true' or coalesce(with_check, '') = 'true'
  union all
  -- 4. Представление без security_invoker читает таблицы правами владельца и обходит RLS
  select 4, 'представление обходит RLS', c.relname, 'через него видны данные всех клубов'
  from pg_class c join pg_namespace n on n.oid = c.relnamespace
  where n.nspname = 'public' and c.relkind in ('v', 'm')
    and not coalesce((select o.option_value::boolean from pg_options_to_table(c.reloptions) o
                      where o.option_name = 'security_invoker'), false)
  union all
  -- 5. Таблица с данными игроков без org_id: политику по клубу на неё не написать
  select 5, 'нет org_id', table_name, 'нельзя ограничить по клубу без лишних join'
  from information_schema.columns where table_schema = 'public'
  group by table_name having bool_or(column_name = 'player_id') and not bool_or(column_name = 'org_id')
  union all
  -- 6. SECURITY DEFINER в public вызывается через API (rpc): внутри должна быть проверка вызывающего
  select 6, 'функция с правами владельца', p.proname,
         case when pg_get_functiondef(p.oid) ~* 'auth\.uid\(\)|is_member|federation_of' then 'проверка вызывающего есть — сверить логику'
              else 'нет проверки вызывающего: отдаёт данные любому' end
  from pg_proc p join pg_namespace n on n.oid = p.pronamespace
  where n.nspname = 'public' and p.prosecdef
) r
order by n, obekt;
izolyaciya.sql — Живые проверки под пользователями: тренер А, тренер Б, аналитик федерации, не вошедший. Пять значений в начале — ваши id
-- Живая проверка изоляции клубов: заходим под реальными пользователями (роль authenticated + их sub в JWT),
-- пробуем прочитать и изменить чужое, смотрим сводку федерации и что видит не вошедший (anon).
-- Ничего не портит: каждая запись откатывается внутри своего блока. Запуск: SQL Editor на КОПИИ проекта, роль postgres.
-- Настройка — пять значений ниже: тренер клуба А, тренер клуба Б, аналитик федерации, id клубов А и Б.
drop table if exists pg_temp.rls_itog;
create temp table rls_itog (n serial, kto text, proverka text, ok boolean, fakt text);
grant all on rls_itog to public;

create or replace function pg_temp.kak(p_uid uuid) returns void language plpgsql as $$
begin
  perform set_config('request.jwt.claims',
    json_build_object('sub', p_uid, 'role', case when p_uid is null then 'anon' else 'authenticated' end)::text, true);
  perform set_config('role', case when p_uid is null then 'anon' else 'authenticated' end, true);
end $$;

create or replace function pg_temp.itog(p_kto text, p_proverka text, p_ok boolean, p_fakt text) returns void language plpgsql as $$
begin
  perform set_config('role', 'none', true);          -- результат пишем своими правами
  insert into rls_itog (kto, proverka, ok, fakt) values (p_kto, p_proverka, p_ok, p_fakt);
end $$;

do $$
declare
  -- ── настройка под проект ──
  coach_a  uuid := '11111111-1111-1111-1111-11111111111a';
  coach_b  uuid := '22222222-2222-2222-2222-22222222222b';
  analyst  uuid := '33333333-3333-3333-3333-33333333333f';
  club_a   uuid := '00000000-0000-0000-0000-0000000000a1';
  club_b   uuid := '00000000-0000-0000-0000-0000000000b1';
  -- ──
  n int; m int; t text; imya_b text;
begin
  select name into imya_b from organizations where id = club_b;   -- имя клуба Б берём своими правами: тренеру А оно не видно
  -- Тренер клуба А: своё видит, чужое — нет
  perform pg_temp.kak(coach_a); select count(*) into n from players where org_id = club_a;
  perform pg_temp.itog('тренер А', 'видит своих игроков', n > 0, n || ' строк');
  foreach t in array array['players', 'ratings', 'tournaments', 'dev_plans'] loop
    perform pg_temp.kak(coach_a);
    execute format('select count(*) from %I where org_id = $1', t) into n using club_b;
    perform pg_temp.itog('тренер А', 'не видит ' || t || ' клуба Б', n = 0, n || ' строк клуба Б');
  end loop;
  perform pg_temp.kak(coach_a); select count(*) into n from ratings_board where club = imya_b;
  perform pg_temp.itog('тренер А', 'витрина ratings_board без строк клуба Б', n = 0, n || ' строк клуба Б');

  -- запись в чужой клуб: вставка, правка, удаление (всё откатывается)
  begin
    perform pg_temp.kak(coach_a);
    insert into players (org_id, full_name) values (club_b, 'подброшенный игрок');
    raise exception 'ОТКАТ 1';
  exception when others then
    if sqlerrm like 'ОТКАТ%' then perform pg_temp.itog('тренер А', 'не может добавить игрока в клуб Б', false, 'вставка прошла');
    else perform pg_temp.itog('тренер А', 'не может добавить игрока в клуб Б', true, 'отказ: ' || sqlerrm); end if;
  end;
  begin
    perform pg_temp.kak(coach_a);
    update ratings set points = points + 500 where org_id = club_b; get diagnostics n = row_count;
    raise exception 'ОТКАТ %', n;
  exception when others then
    if sqlerrm like 'ОТКАТ%' then m := split_part(sqlerrm, ' ', 2)::int;
      perform pg_temp.itog('тренер А', 'не может менять рейтинги клуба Б', m = 0, m || ' строк изменено');
    else perform pg_temp.itog('тренер А', 'не может менять рейтинги клуба Б', true, 'отказ: ' || sqlerrm); end if;
  end;
  begin
    perform pg_temp.kak(coach_a);
    update players set org_id = club_a where org_id = club_b; get diagnostics n = row_count;
    raise exception 'ОТКАТ %', n;
  exception when others then
    if sqlerrm like 'ОТКАТ%' then m := split_part(sqlerrm, ' ', 2)::int;
      perform pg_temp.itog('тренер А', 'не может переманить игрока Б к себе', m = 0, m || ' строк изменено');
    else perform pg_temp.itog('тренер А', 'не может переманить игрока Б к себе', true, 'отказ: ' || sqlerrm); end if;
  end;

  -- Тренер клуба Б зеркально
  perform pg_temp.kak(coach_b); select count(*) into n from players where org_id = club_a;
  perform pg_temp.itog('тренер Б', 'не видит игроков клуба А', n = 0, n || ' строк клуба А');

  -- Федерация: сводка по своим клубам есть, ФИО и телефонов игроков нет
  perform pg_temp.kak(analyst); select count(*) into n from players;
  perform pg_temp.itog('федерация', 'не видит личные данные игроков', n = 0, n || ' строк players');
  perform pg_temp.kak(analyst); select count(*) into n from federation_summary();
  perform pg_temp.itog('федерация', 'видит сводку по обоим клубам', n = 2, n || ' клубов в сводке');
  perform pg_temp.kak(coach_a); select count(*) into n from federation_summary();
  perform pg_temp.itog('тренер А', 'не получает сводку федерации', n = 0, n || ' строк');

  -- Не вошедший (anon)
  foreach t in array array['players', 'ratings', 'dev_plans', 'ratings_board'] loop
    perform pg_temp.kak(null);
    begin
      execute format('select count(*) from %I', t) into n;
    exception when insufficient_privilege then n := 0;
    end;
    perform pg_temp.itog('не вошёл', 'не видит ' || t, n = 0, n || ' строк');
  end loop;
  perform set_config('role', 'none', true);
end $$;

-- Исправления четырёх дыр модели. На реальном проекте — миграцией, после неё снова audit.sql и izolyaciya.sql.

-- 1. dev_plans: включить RLS и пустить только свой клуб (тренеры и админы)

model.sql — Модель по описанию проекта, на которой сделан прогон
-- Модель по описанию проекта (НЕ код заказчика): федерация → клубы → игроки, рейтинги, турниры, планы развития.
-- Окружение как в Supabase: роли anon / authenticated / service_role, auth.uid() читает sub из request.jwt.claims.
-- В модель специально заложены 4 типичные ошибки — их должны поймать audit.sql и izolyaciya.sql.

-- ── окружение Supabase ──
do $$ begin
  if not exists (select 1 from pg_roles where rolname = 'anon') then create role anon nologin; end if;
  if not exists (select 1 from pg_roles where rolname = 'authenticated') then create role authenticated nologin; end if;
  if not exists (select 1 from pg_roles where rolname = 'service_role') then create role service_role nologin bypassrls; end if;
end $$;
create schema if not exists auth;
create or replace function auth.uid() returns uuid language sql stable as
$$ select nullif(current_setting('request.jwt.claims', true)::json ->> 'sub', '')::uuid $$;
grant usage on schema auth, public to anon, authenticated, service_role;
grant execute on function auth.uid() to anon, authenticated, service_role;

-- ── схема ──
create table organizations (
  id uuid primary key, name text not null,
  kind text not null check (kind in ('federation', 'club')),
  parent_id uuid references organizations(id)            -- клуб → его федерация
);
create table memberships (                                -- кто в какой организации и с какой ролью
  user_id uuid not null, org_id uuid not null references organizations(id),
  role text not null check (role in ('admin', 'coach', 'analyst')),
  primary key (user_id, org_id)
);
create table players (
  id uuid primary key default gen_random_uuid(), org_id uuid not null references organizations(id),
  full_name text not null, birth_date date, phone text      -- личные данные: федерации видеть не нужно
);
create table ratings (
  player_id uuid not null references players(id), org_id uuid not null references organizations(id),
  points int not null, updated_at date not null default current_date
);
create table tournaments (
  id uuid primary key default gen_random_uuid(), org_id uuid not null references organizations(id),
  title text not null, starts_on date
);
create table dev_plans (                                  -- планы развития игроков
  player_id uuid not null references players(id), org_id uuid not null references organizations(id),
  notes text
);
grant select, insert, update, delete on all tables in schema public to anon, authenticated;   -- так по умолчанию делает Supabase

-- ── помощники для политик ──
create or replace function is_member(p_org uuid, p_roles text[] default array['admin','coach','analyst'])
returns boolean language sql stable security definer set search_path = public as
$$ select exists (select 1 from memberships m where m.user_id = auth.uid() and m.org_id = p_org and m.role = any(p_roles)) $$;

create or replace function federation_of(p_club uuid) returns boolean language sql stable security definer set search_path = public as
$$ select exists (select 1 from organizations c join memberships m on m.org_id = c.parent_id
                  where c.id = p_club and m.user_id = auth.uid()) $$;

-- ── RLS ──
alter table organizations enable row level security;
alter table memberships   enable row level security;
alter table players       enable row level security;
alter table ratings       enable row level security;
alter table tournaments   enable row level security;
-- ОШИБКА 1: на dev_plans RLS забыли включить

create policy org_read on organizations for select to authenticated using (is_member(id) or federation_of(id));
create policy mem_read on memberships for select to authenticated using (user_id = auth.uid());

create policy players_read   on players for select to authenticated using (is_member(org_id));
create policy players_update on players for update to authenticated using (is_member(org_id, array['admin','coach']))
                                                                      with check (is_member(org_id, array['admin','coach']));
-- ОШИБКА 2: вставка без проверки клуба — тренер может завести игрока в чужой клуб
create policy players_insert on players for insert to authenticated with check (true);

create policy ratings_read  on ratings for select to authenticated using (is_member(org_id));
create policy ratings_write on ratings for update to authenticated using (is_member(org_id, array['admin','coach']))
                                                                  with check (is_member(org_id, array['admin','coach']));
-- ОШИБКА 3: турниры «для всех» — вместе с внутренними турнирами чужих клубов
create policy tournaments_read on tournaments for select to authenticated using (true);

-- ОШИБКА 4: витрина рейтинга без security_invoker — читает таблицы правами владельца и обходит RLS
create view ratings_board as
  select p.full_name, o.name as club, r.points from ratings r join players p on p.id = r.player_id join organizations o on o.id = r.org_id;
grant select on ratings_board to anon, authenticated;

-- Сводка для федерации: только агрегаты по своим клубам, без ФИО и телефонов (так и задумано)
create or replace function federation_summary()
returns table (club text, players bigint, avg_points numeric) language sql stable security definer set search_path = public as
$$ select o.name, count(distinct p.id), round(avg(r.points), 1)
   from organizations o left join players p on p.org_id = o.id left join ratings r on r.player_id = p.id
   where o.kind = 'club' and federation_of(o.id)
   group by o.name order by o.name $$;
revoke execute on function federation_summary() from public;
grant execute on function federation_summary() to authenticated;
